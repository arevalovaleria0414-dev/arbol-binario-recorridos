import time # ----------------------------- # CLASE NODO # ----------------------------- class Nodo: def __init__(self, valor): self.valor = valor self.izquierda = None self.derecha = None # ----------------------------- # CLASE ARBOL BINARIO # ----------------------------- class ArbolBinario: def __init__(self): self.raiz = None # ----------------------------- # INSERTAR NODOS # ----------------------------- def insertar(self, valor): print(f"\nInsertando {valor} en el árbol...") if self.raiz is None: self.raiz = Nodo(valor) print(f"{valor} se convierte en la RAÍZ del árbol") else: self._insertar_recursivo(self.raiz, valor) def _insertar_recursivo(self, nodo, valor): print(f"Comparando {valor} con nodo {nodo.valor}") time.sleep(1) if valor < nodo.valor: print(f"{valor} es MENOR que {nodo.valor} → ir a la IZQUIERDA") time.sleep(1) if nodo.izquierda is None: nodo.izquierda = Nodo(valor) print(f"Se creó el nodo {valor} a la izquierda de {nodo.valor}") else: self._insertar_recursivo(nodo.izquierda, valor) else: print(f"{valor} es MAYOR que {nodo.valor} → ir a la DERECHA") time.sleep(1) if nodo.derecha is None: nodo.derecha = Nodo(valor) print(f"Se creó el nodo {valor} a la derecha de {nodo.valor}") else: self._insertar_recursivo(nodo.derecha, valor) # ----------------------------- # MOSTRAR ARBOL # ----------------------------- def mostrar_arbol(self, nodo, nivel=0): if nodo is not None: self.mostrar_arbol(nodo.derecha, nivel + 1) print(" " * nivel + "→", nodo.valor) self.mostrar_arbol(nodo.izquierda, nivel + 1) # ----------------------------- # RECORRIDO INORDER VISUAL # ----------------------------- def inorder_visual(self, nodo): if nodo: print(f"\n⬅️ Ir a la izquierda de {nodo.valor}") time.sleep(1) self.inorder_visual(nodo.izquierda) print(f"🟢 VISITANDO nodo {nodo.valor}") time.sleep(1) print(f"➡️ Ir a la derecha de {nodo.valor}") time.sleep(1) self.inorder_visual(nodo.derecha) # ----------------------------- # RECORRIDO PREORDER VISUAL # ----------------------------- def preorder_visual(self, nodo): if nodo: print(f"\n🟢 VISITANDO nodo {nodo.valor}") time.sleep(1) print(f"⬅️ Ir a la izquierda de {nodo.valor}") time.sleep(1) self.preorder_visual(nodo.izquierda) print(f"➡️ Ir a la derecha de {nodo.valor}") time.sleep(1) self.preorder_visual(nodo.derecha) # ----------------------------- # RECORRIDO POSTORDER VISUAL # ----------------------------- def postorder_visual(self, nodo): if nodo: print(f"\n⬅️ Ir a la izquierda de {nodo.valor}") time.sleep(1) self.postorder_visual(nodo.izquierda) print(f"➡️ Ir a la derecha de {nodo.valor}") time.sleep(1) self.postorder_visual(nodo.derecha) print(f"🟢 VISITANDO nodo {nodo.valor}") time.sleep(1) # ----------------------------- # PROGRAMA PRINCIPAL # ----------------------------- arbol = ArbolBinario() print("\n========== CREACIÓN DEL ÁRBOL ==========") cantidad = int(input("\n¿Cuántos números desea ingresar?: ")) for i in range(cantidad): numero = int(input(f"Ingrese número {i+1}: ")) arbol.insertar(numero) print("\n========== ÁRBOL BINARIO RESULTANTE ==========\n") arbol.mostrar_arbol(arbol.raiz) print("\n========== RECORRIDO INORDER ==========") arbol.inorder_visual(arbol.raiz) print("\n========== RECORRIDO PREORDER ==========") arbol.preorder_visual(arbol.raiz) print("\n========== RECORRIDO POSTORDER ==========") arbol.postorder_visual(arbol.raiz) print("\n\nFIN DEL PROGRAMA")
